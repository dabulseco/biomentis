def predict_binding_score(nb_props, epitope_props):
    """
    Predict binding affinity based on:

    2. Electrostatic complementarity (charge matching)
    3. Hydrophobic matching
    4. Aromatic interactions
    """

    # 1. CDR3 length score (longer CDR3 = better for concave epitopes)
    cdr3_len = nb_props['CDR3_length']
    if 12 <= cdr3_len <= 20:
        cdr3_score = 1.0
    elif 10 <= cdr3_len < 12:
        cdr3_score = 0.8
    elif cdr3_len > 20:
        cdr3_score = 0.7
    else:
        cdr3_score = 0.5
    scores['CDR3_length'] = cdr3_score
    
    # 2. Stability score (instability index < 40 is good)
    instability = nb_props['instability_index']
    if instability < 20:
        stability_score = 1.0
    elif instability < 30:
        stability_score = 0.8
    elif instability < 40:
        stability_score = 0.6
    else:
        stability_score = 0.3
    scores['stability'] = stability_score
    
    # 3. Hydrophilicity score (GRAVY < 0 is good for solubility)
    gravy = nb_props['GRAVY']
    if -0.6 < gravy < -0.2:
        gravy_score = 1.0
    elif -0.8 < gravy <= -0.6 or -0.2 <= gravy < 0:
        gravy_score = 0.8
    else:
        gravy_score = 0.5
    scores['solubility'] = gravy_score
    
    # 4. Aromatic content (important for epitope recognition)
    aromatic = nb_props['aromatic_content']
    if 0.12 <= aromatic <= 0.20:
        aromatic_score = 1.0
    elif 0.08 <= aromatic < 0.12 or 0.20 < aromatic <= 0.25:
        aromatic_score = 0.8
    else:
        aromatic_score = 0.5
    scores['aromatic'] = aromatic_score
    
    # 5. Net charge (moderate charge preferred)
    charge = abs(nb_props['net_charge_pH7.4'])
    if charge <= 2:
        charge_score = 1.0
    elif charge <= 4:
        charge_score = 0.8
    else:
        charge_score = 0.5
    scores['charge'] = charge_score
    
    # Weighted total
    weights = {
        'CDR3_length': 0.30,
        'stability': 0.25,
        'solubility': 0.20,
        'aromatic': 0.15,
        'charge': 0.10

    }

def predict_cross_reactivity(nb_name, target_epitope, base_dG):
    """
    Predict binding affinity across genotypes based on epitope conservation.
    More conserved epitopes = better cross-reactivity.
    """
    results = {}
    
    # Extract epitope name
    if 'Domain_B_arch2' in target_epitope:
        epitope_key = 'Domain_B_arch2'
    elif 'Domain_A_lateral' in target_epitope:
        epitope_key = 'Domain_A_lateral'
    elif 'Domain_A_central' in target_epitope:
        epitope_key = 'Domain_A_central'
    else:
        epitope_key = 'Domain_B_arch2'
    
    dG_values = []
    for genotype, data in genotype_variations.items():
        conservation = data[epitope_key]['conservation']
        mutations = data[epitope_key]['mutations']
        
        # Penalty for each mutation in the epitope
        mutation_penalty = len(mutations) * 0.5  # kcal/mol per mutation
        
        # Adjusted dG based on conservation
        if conservation >= 0.95:
            dG_adjustment = 0
        elif conservation >= 0.90:
            dG_adjustment = 0.5
        elif conservation >= 0.85:
            dG_adjustment = 1.0
        else:
            dG_adjustment = 2.0
        
        genotype_dG = base_dG + dG_adjustment + mutation_penalty
        dG_values.append(genotype_dG)
        
        results[genotype] = {
            'conservation': conservation,
            'mutations': mutations,
            'predicted_dG': round(genotype_dG, 1),
            'dG_shift': round(genotype_dG - base_dG, 1)
        }
    
    # Cross-reactivity metrics
    dG_array = np.array(dG_values)
    results['mean_dG'] = round(np.mean(dG_array), 1)
    results['std_dG'] = round(np.std(dG_array), 2)
    results['max_shift'] = round(np.max(dG_array) - np.min(dG_array), 1)
    
    # Cross-reactivity score (lower std = better)
    if results['std_dG'] < 0.5:
        results['cross_reactivity'] = 'Excellent'
        results['cross_reactivity_score'] = 1.0
    elif results['std_dG'] < 1.0:
        results['cross_reactivity'] = 'Good'
        results['cross_reactivity_score'] = 0.8
    elif results['std_dG'] < 2.0:
        results['cross_reactivity'] = 'Moderate'
        results['cross_reactivity_score'] = 0.6
    else:
        results['cross_reactivity'] = 'Poor'
        results['cross_reactivity_score'] = 0.3
    
    return results
